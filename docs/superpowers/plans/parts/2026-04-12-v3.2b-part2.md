```python
from __future__ import annotations
import logging
from dataclasses import dataclass, field
from pathlib import Path
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from neoagent.agent import NeoAgent
    from neoagent.multi.orchestrator import Orchestrator

logger = logging.getLogger(__name__)


# ── WorkerCard ────────────────────────────────────────────────────────────────

@dataclass(frozen=True)
class WorkerCard:
    name: str
    description: str
    instruction: str
    tags: tuple[str, ...] = ()
    model: str | None = None
    tools: tuple[str, ...] = ()


# ── WorkerPool ────────────────────────────────────────────────────────────────

class WorkerPool:
    def __init__(self) -> None:
        self._cards: dict[str, WorkerCard] = {}

    def register(self, card: WorkerCard) -> None:
        if card.name in self._cards:
            logger.warning("WorkerPool: overwriting existing worker %r", card.name)
        self._cards[card.name] = card

    def find(self, name: str) -> WorkerCard | None:
        return self._cards.get(name)

    def list_all(self) -> list[WorkerCard]:
        return list(self._cards.values())

    def remove(self, name: str) -> bool:
        if name in self._cards:
            del self._cards[name]
            return True
        return False


# ── Worker factory ────────────────────────────────────────────────────────────

def _create_worker_agent(
    card: WorkerCard,
    orchestrator: "Orchestrator",
    depth: int,
) -> "NeoAgent":
    """Create an independent NeoAgent instance from a WorkerCard.

    Registers only tools explicitly authorized by card.tools.
    Registers spawn_worker only if depth < orchestrator.max_depth.
    Sets up event bubbling to the Orchestrator's EventBus.
    """
    from neoagent.agent import NeoAgent
    from neoagent.config import NeoAgentConfig

    config = NeoAgentConfig(
        api_key=orchestrator.config.api_key,
        model=card.model or orchestrator.config.model,
        provider=orchestrator.config.provider,
        base_url=orchestrator.config.base_url,
        system_prompt=card.instruction,
        auto_approve_tools=True,  # internal workers auto-approve
    )
    agent = NeoAgent(config)

    # Register only tools authorized by WorkerCard
    for tool_name in card.tools:
        tool = orchestrator._tool_pool.get(tool_name)
        if tool is not None:
            agent.register_tool(tool)
        else:
            logger.debug(
                "_create_worker_agent: tool %r not found in pool, skipping", tool_name
            )

    # Depth guard: only register spawn_worker if depth < max_depth
    if depth < orchestrator.max_depth:
        from neoagent.multi.tools.spawn_worker import SpawnWorkerTool
        spawn_tool = SpawnWorkerTool(orchestrator=orchestrator, depth=depth)
        agent.register_tool(spawn_tool)

    # Event bubbling: worker events → orchestrator event bus
    from neoagent.multi.events import _setup_event_bubble
    _setup_event_bubble(agent, orchestrator, card.name, depth)

    return agent
```

```python
    agent: "NeoAgent",
    task: "Task",
    tracker: "TaskTracker",

    """Execute a worker task. Handles timeout and cancellation gracefully."""
    timeout = task.constraints.get("timeout", 1800)
    max_turns = task.constraints.get("max_turns", 20)

    from neoagent.core.types import Message as _Message

    # Build message list: context + instruction
    messages = list(task.context) + [_Message(role="user", content=task.instruction)]

    try:
        async with asyncio.timeout(timeout):
            result = await agent.run(messages=messages)

        # Extract output from last assistant turn
        from neoagent.core.types import TextBlock as _TB
        output = ""
        if result.turns:
            last = result.turns[-1].response
            if isinstance(last.content, list):
                output = "\n".join(
                    b.text for b in last.content if isinstance(b, _TB)
                )
            elif isinstance(last.content, str):
                output = last.content

        # Truncate output to max_output_tokens (approximate: 4 chars/token)
        max_chars = task.max_output_tokens * 4
        if len(output) > max_chars:
            output = output[:max_chars] + "\n[truncated]"

        # Accumulate token usage across all turns
        total_input = sum(
            getattr(t.response, "input_tokens", 0) for t in result.turns
        )
        total_output = sum(
            getattr(t.response, "output_tokens", 0) for t in result.turns
        )
        # Prefer provider-level usage from session state
        usage = TokenUsage(input_tokens=total_input, output_tokens=total_output)

        return TaskResult(
            task_id=task.task_id,
            status="completed",
            output=output,
            error=None,
            usage=usage,
            work_summary=None,
            turns_completed=len(result.turns),
        )

    except asyncio.CancelledError:
        summary = _extract_work_summary(agent)
        return TaskResult(
            task_id=task.task_id,
            status="cancelled",
            output="",
            error=None,
            usage=None,
            work_summary=summary,
            turns_completed=0,
        )
    except Exception as e:
        return TaskResult(
            task_id=task.task_id,
            status="failed",
            output="",
            error=str(e),
            usage=None,
            work_summary=None,
```

```python
class SpawnWorkerTool(BaseTool):
    """Dynamically spawn a temporary Worker from a markdown definition and run a task.

    The WorkerCard is NOT registered to WorkerPool — it is ephemeral.
    Parses md_definition → WorkerCard → NeoAgent → runs task → returns TaskResult.
    """
    name: str = "spawn_worker"
    description: str = (
        "Dynamically create a temporary Worker from a markdown definition and run a task. "
        "The md_definition must be a valid WorkerCard .md format with a 'name' field. "
        "The worker is ephemeral — not added to the worker pool."
    )
    input_model: type[BaseModel] = SpawnWorkerInput
    permission: str = "auto"

    def __init__(self, orchestrator: "Orchestrator", depth: int) -> None:
        self._orchestrator = orchestrator
        self._depth = depth

    async def execute(self, input: BaseModel) -> ToolResult:
        assert isinstance(input, SpawnWorkerInput)

        # 1. Parse md_definition → WorkerCard
        from neoagent.multi.parser import parse_worker_md
        from neoagent.multi.task import Task, _run_worker
        from neoagent.multi.worker import _create_worker_agent

        try:
            card = parse_worker_md(input.md_definition)
        except ValueError as e:
            return ToolResult(
                call_id="",
                output=f"Failed to parse worker definition: {e}",
                is_error=True,
            )

        # 2. Create worker agent (depth + 1 for spawned workers)
        task_id = str(uuid.uuid4())[:8]
        worker_agent = _create_worker_agent(card, self._orchestrator, self._depth + 1)

        # 3. Build Task
        task = Task(
            task_id=task_id,
            instruction=input.instruction,
        )

        # 4. Execute under semaphore
        async with self._orchestrator._semaphore:
            result = await _run_worker(worker_agent, task, self._orchestrator._task_tracker)

        # 5. Track completion
        self._orchestrator._task_tracker.complete(task_id, result)
```

```python
from __future__ import annotations
import asyncio
import logging
from pathlib import Path
from typing import TYPE_CHECKING

from neoagent.config import NeoAgentConfig
from neoagent.agent import NeoAgent
from neoagent.multi.worker import WorkerPool
from neoagent.multi.task import TaskTracker
from neoagent.tools.base import BaseTool

if TYPE_CHECKING:
    pass

logger = logging.getLogger(__name__)


def _create_provider(config: NeoAgentConfig):
    """Re-export to allow patching in tests."""
    from neoagent.agent import _create_provider as _cp
    return _cp(config)


class Orchestrator:
    """Multi-agent coordinator.

    Composes a NeoAgent (brain) with WorkerPool and TaskTracker.
    The brain's LLM has access to 5 internal tools for spawning and delegating tasks.
    Workers are independent NeoAgent instances created on demand.

    Usage::

        orch = Orchestrator(NeoAgentConfig(api_key="..."))
        orch.register_worker(WorkerCard(name="reviewer", ...))
        orch.register_tool(ReadTool())
        result = await orch.run("Review these three files for security issues")
    """

    def __init__(
        self,
        config: NeoAgentConfig,
        max_depth: int = 2,
        max_concurrent_workers: int = 5,
    ) -> None:
        self.config = config
        self.max_depth = max_depth
        self.max_concurrent_workers = max_concurrent_workers

        # Internal components
        self._worker_pool = WorkerPool()
        self._task_tracker = TaskTracker()
        self._tool_pool: dict[str, BaseTool] = {}
        self._semaphore = asyncio.Semaphore(max_concurrent_workers)

        # Brain: the orchestrator's own NeoAgent
        self._brain = NeoAgent(config)

        # Register 5 internal tools onto the brain
        self._register_internal_tools()

    def _register_internal_tools(self) -> None:
        from neoagent.multi.tools.spawn_worker import SpawnWorkerTool
        from neoagent.multi.tools.delegate_task import DelegateTaskTool
        from neoagent.multi.tools.cancel_task import CancelTaskTool
        from neoagent.multi.tools.list_tools import ListWorkersTool, ListTasksTool

        self._brain.register_tool(SpawnWorkerTool(orchestrator=self, depth=0))
        self._brain.register_tool(DelegateTaskTool(orchestrator=self, depth=0))
        self._brain.register_tool(CancelTaskTool(orchestrator=self))
        self._brain.register_tool(ListWorkersTool(orchestrator=self))
        self._brain.register_tool(ListTasksTool(orchestrator=self))

    # ── Worker registration ───────────────────────────────────────────────────

    def register_worker(self, card) -> None:
        """Register a static WorkerCard by reference."""
        self._worker_pool.register(card)

    def load_workers(self, directory: str | Path) -> None:
        """Load all .md WorkerCard files from a directory and register them."""
        from neoagent.multi.parser import load_workers
        for card in load_workers(Path(directory)):
            self._worker_pool.register(card)

    # ── Tool pool management ──────────────────────────────────────────────────

    def register_tool(self, tool: BaseTool) -> None:
        """Add a tool to the global pool available for Worker authorization."""
        self._tool_pool[tool.name] = tool

    # ── Execution ─────────────────────────────────────────────────────────────

    async def run(self, message: str) -> str:
        """Run the orchestrator with a user message.

        The brain's LLM decides how to decompose and delegate the work.
        Returns the final response string.
        """
        return await self._brain.chat(message)

    # ── Lifecycle ─────────────────────────────────────────────────────────────

    async def close(self) -> None:
        """Cancel all running worker tasks and clean up the brain."""
        # Cancel all tracked active tasks
        active_ids = self._task_tracker.list_active()
        for task_id in active_ids:
            await self._task_tracker.cancel(task_id)

        # Close brain (MCP connections etc.)
        await self._brain.close()

    async def __aenter__(self) -> "Orchestrator":
        return self

    async def __aexit__(self, *_) -> None:
        await self.close()
```

```python
class TaskTracker:
    def __init__(self) -> None:
        self._tasks: dict[str, asyncio.Task] = {}       # task_id → asyncio.Task
        self._results: dict[str, TaskResult] = {}       # task_id → TaskResult
        self._metadata: dict[str, dict] = {}            # task_id → {worker_name, status}

    def track(self, task: Task, async_task: asyncio.Task, worker_name: str = "") -> None:
        """Register a running asyncio.Task for tracking."""
        self._tasks[task.task_id] = async_task
        self._metadata[task.task_id] = {
            "task_id": task.task_id,
            "worker_name": worker_name,
            "status": "running",
        }

    def complete(self, task_id: str, result: TaskResult) -> None:
        """Mark task as complete and store result."""
        self._results[task_id] = result
        self._tasks.pop(task_id, None)
        if task_id in self._metadata:
            self._metadata[task_id]["status"] = result.status

    async def cancel(self, task_id: str) -> bool:
        """Cancel a running task. Returns True if found and cancelled."""
        async_task = self._tasks.get(task_id)
        if async_task is None:
            return False
        async_task.cancel()
        try:
            await asyncio.wait_for(asyncio.shield(async_task), timeout=5.0)
        except (asyncio.CancelledError, asyncio.TimeoutError):
            pass
        self._tasks.pop(task_id, None)
        if task_id in self._metadata:
            self._metadata[task_id]["status"] = "cancelled"
        return True

    def get_result(self, task_id: str) -> TaskResult | None:
        return self._results.get(task_id)

    def list_active(self) -> list[str]:
        return [tid for tid, t in self._tasks.items() if not t.done()]

    def list_all(self) -> list[dict]:
        result = []
        for tid, meta in self._metadata.items():
            result.append(dict(meta))
        return result
```

**Key design choices in Part 2:**

- `_create_worker_agent` is a pure factory function (not a method), matching the spec's design and making it easy to mock in tests
- `system_prompt` is added to `NeoAgentConfig` as `None`-default field — backward compatible, no existing code breaks
- `_run_worker` lives in `task.py` (close to `Task`/`TaskResult` data types), not in a tool file — clean separation
- `_format_task_result` is a shared helper in `tools/__init__.py` — avoids duplication between `spawn_worker` and `delegate_task`
- Event bubbling uses `subscribe_all` on the worker's `EventBus` — wraps every event as `WorkerEvent` without needing to know event types upfront
- `_ALL_EVENT_TYPES` uses lazy import to avoid `events.py` → `multi/events.py` circular dependency
- `CancelTaskTool.execute` is synchronous-friendly — `async def execute` calls `await tracker.cancel()` which handles asyncio task cancellation
- Integration tests mock LLM calls (`agent.run = AsyncMock`) but use real `WorkerPool`, `TaskTracker`, `EventBus`, `Semaphore` — tests the wiring, not the LLM
