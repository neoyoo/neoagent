@dataclass
class Session:
    id: str
    messages: list[Message]
    state: SessionState
    created_at: datetime
    updated_at: datetime
    metadata: dict = field(default_factory=dict)
    _storage: "SessionStorage | None" = field(default=None, repr=False)

    @classmethod
    def create(cls, session_id: str | None = None) -> "Session":
        now = datetime.now(timezone.utc).replace(tzinfo=None)
        return cls(
            id=session_id or str(uuid.uuid4()),
            messages=[],
            state=SessionState(),
            created_at=now,
            updated_at=now,
        )

    @classmethod
    def resume(cls, session_id: str, storage: "SessionStorage") -> "Session":
        return storage.load(session_id)

    def save(self, storage: "SessionStorage") -> None:
        self.updated_at = datetime.now(timezone.utc).replace(tzinfo=None)
        storage.save(self)

    def save_if_storage(self) -> None:
        """如果绑定了 storage，执行 save；否则静默跳过。"""
        if self._storage is not None:
            self.save(self._storage)

    def bind_storage(self, storage: "SessionStorage") -> None:
        """绑定 storage，之后 save_if_storage() 生效。"""
        self._storage = storage

    def fork(self, new_id: str | None = None) -> "Session":
        now = datetime.now(timezone.utc).replace(tzinfo=None)
        return Session(
            id=new_id or str(uuid.uuid4()),
            messages=copy.deepcopy(self.messages),
            state=copy.deepcopy(self.state),
            created_at=now,
            updated_at=now,
            metadata=copy.deepcopy(self.metadata),
        )
```

    def cleanup(
        self,
        max_age_days: int = 30,
        max_sessions: int = 100,
    ) -> list[str]:
        """清理过期或超量的 session 文件。

        Pass 1：删除修改时间超过 max_age_days 天的文件。
        Pass 2：若剩余文件数仍超过 max_sessions，删除最旧的直到满足数量约束。

        Returns:
            被清理的 session_id 列表（按删除顺序排列）。
        """
        import time
        now = time.time()
        max_age_secs = max_age_days * 86400
        paths = sorted(self._base_dir.glob("*.json"), key=lambda p: p.stat().st_mtime)

        removed: list[str] = []

        # Pass 1: 删除超龄文件
        surviving = []
        for p in paths:
            if now - p.stat().st_mtime > max_age_secs:
                p.unlink(missing_ok=True)
                removed.append(p.stem)
            else:
                surviving.append(p)

        # Pass 2: 超量时删除最旧的（surviving 已按 mtime 升序排列）
        if len(surviving) > max_sessions:
            to_remove = surviving[: len(surviving) - max_sessions]
            for p in to_remove:
                p.unlink(missing_ok=True)
                removed.append(p.stem)

        return removed
```

# neoagent/eval/usage.py
from __future__ import annotations

from dataclasses import dataclass, field
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from neoagent.events import ProviderResponseEvent


@dataclass
class ModelUsage:
    """单个模型的 token 用量汇总。"""
    model: str
    input_tokens: int = 0
    output_tokens: int = 0
    request_count: int = 0


class UsageTracker:
    """追踪 agent 运行期间的 API 调用 token 用量（纯统计，不计算金额）。

    通过订阅 ProviderResponseEvent 自动收集数据。

    Usage::

        tracker = agent.enable_usage_tracking()
        await agent.chat("hello", session=session)
        print(f"Total input tokens: {tracker.total_input_tokens}")
        print(f"Total output tokens: {tracker.total_output_tokens}")
    """

    def __init__(self) -> None:
        self._per_model: dict[str, ModelUsage] = {}
        self._current_model: str = "unknown"

    def _handle_response(self, event: "ProviderResponseEvent") -> None:
        """EventBus handler，由 enable_usage_tracking() 注册。

        _current_model 在 enable_usage_tracking() 中绑定（provider.model）。
        """
        model = self._current_model
        if model not in self._per_model:
            self._per_model[model] = ModelUsage(model=model)
        usage = self._per_model[model]
        usage.input_tokens += event.input_tokens
        usage.output_tokens += event.output_tokens
        usage.request_count += 1

    @property
    def total_input_tokens(self) -> int:
        """所有模型的输入 token 总量。"""
        return sum(u.input_tokens for u in self._per_model.values())

    @property
    def total_output_tokens(self) -> int:
        """所有模型的输出 token 总量。"""
        return sum(u.output_tokens for u in self._per_model.values())

    @property
    def per_model_usage(self) -> dict[str, ModelUsage]:
        """按模型分组的用量统计（返回副本）。"""

# neoagent/eval/metrics.py
from __future__ import annotations

import time
from dataclasses import dataclass, field
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from neoagent.events import EventBus


@dataclass
class TurnMetrics:
    """单轮次的指标快照。"""
    turn_index: int
    input_tokens: int
    output_tokens: int
    latency_ms: float           # ProviderRequestEvent → ProviderResponseEvent 耗时
    tool_call_count: int
    tool_names: list[str]       # 本轮调用的工具名列表（按调用顺序）


@dataclass
class SessionMetrics:
    """整个 session（或 eval case）的汇总指标。"""
    turns: list[TurnMetrics] = field(default_factory=list)

    @property
    def total_input_tokens(self) -> int:
        return sum(t.input_tokens for t in self.turns)

    @property
    def total_output_tokens(self) -> int:
        return sum(t.output_tokens for t in self.turns)

    @property
    def total_duration_ms(self) -> float:
        return sum(t.latency_ms for t in self.turns)

    @property
    def total_tool_calls(self) -> int:
        return sum(t.tool_call_count for t in self.turns)


class MetricsCollector:
    """收集 agent 运行期间的 per-turn 和 session 级指标。

    订阅 EventBus 的以下事件：
    - ProviderRequestEvent：记录轮次开始时间
    - ProviderResponseEvent：记录 token 数，计算延迟
    - ToolCallEvent：累计工具调用名称
    - TurnCompleteEvent：封装 TurnMetrics 并追加到 turns 列表

    Usage::

        collector = agent.enable_metrics()
        await agent.chat("hello", session=session)
        metrics = collector.get_session_metrics()
        print(f"Total turns: {len(metrics.turns)}")
    """

    def __init__(self) -> None:
        self._pending: dict[int, dict] = {}   # turn_index → 累计中间数据
        self._turns: list[TurnMetrics] = []

    def _on_provider_request(self, event) -> None:
        self._pending[event.turn] = {
            "start_ms": time.monotonic() * 1000,
            "input_tokens": 0,
            "output_tokens": 0,
            "tool_names": [],
        }

    def _on_provider_response(self, event) -> None:
        p = self._pending.setdefault(event.turn, {})
        p["input_tokens"] = event.input_tokens
        p["output_tokens"] = event.output_tokens
        start = p.get("start_ms")
        p["latency_ms"] = (time.monotonic() * 1000 - start) if start else 0.0

    def _on_tool_call(self, event) -> None:
        # ToolCallEvent 不含 turn_index；归入当前最新未完成轮
        if self._pending:
            latest_turn = max(self._pending.keys())
            self._pending[latest_turn].setdefault("tool_names", []).append(event.name)

    def _on_turn_complete(self, event) -> None:
        p = self._pending.pop(event.turn_index, {})
        tm = TurnMetrics(
            turn_index=event.turn_index,
            input_tokens=p.get("input_tokens", 0),
            output_tokens=p.get("output_tokens", 0),
            latency_ms=p.get("latency_ms", 0.0),
            tool_call_count=event.tool_call_count,
            tool_names=p.get("tool_names", []),
        )
        self._turns.append(tm)

    def get_session_metrics(self) -> SessionMetrics:
        """返回当前收集到的全量 session 指标。"""
        return SessionMetrics(turns=list(self._turns))

    def get_turn_metrics(self, turn_index: int) -> TurnMetrics | None:
        """返回指定 turn_index 的指标，不存在返回 None。"""
        for tm in self._turns:
            if tm.turn_index == turn_index:
                return tm
        return None

    def reset(self) -> None:
        """清空所有数据，用于 EvalRunner 的 per-case 复用。"""
        self._pending.clear()
        self._turns.clear()
```

# neoagent/eval/runner.py
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Callable, TYPE_CHECKING

if TYPE_CHECKING:
    from neoagent.agent import NeoAgent
    from neoagent.core.types import ConversationResult, Message
    from neoagent.eval.metrics import SessionMetrics


@dataclass
class EvalCase:
    """单个评测用例。

    Attributes:
        name:      用例名称，唯一标识符（用于报告）。
        messages:  输入消息列表（将传入 agent.run()）。
        assertion: 断言函数，签名 (ConversationResult) -> bool。
                   True = pass，False = fail。
        max_turns: 覆盖默认的 max_turns（可选）。
        metadata:  用户自定义透传数据（标签、期望答案等）。
    """
    name: str
    messages: list["Message"]
    assertion: Callable[["ConversationResult"], bool]
    max_turns: int | None = None
    metadata: dict = field(default_factory=dict)


@dataclass
class EvalCaseResult:
    """单个 EvalCase 的执行结果。"""
    name: str
    passed: bool
    result: "ConversationResult | None"    # agent 成功执行时有值
    error: str | None                      # agent 抛异常时有值
    metrics: "SessionMetrics | None"       # 若启用了 MetricsCollector 则有值


@dataclass
class EvalReport:
    """批量评测报告。"""
    total: int
    passed: int
    failed: int
    cases: list[EvalCaseResult] = field(default_factory=list)

    @property
    def pass_rate(self) -> float:
        """通过率（passed / total），total == 0 时返回 0.0。"""
        return self.passed / self.total if self.total > 0 else 0.0


class EvalRunner:
    """批量执行 EvalCase 列表，输出 EvalReport。

    每个 case 使用独立的 MetricsCollector 实例，互不干扰。
    agent 实例在 case 之间复用（共享工具注册、provider 配置）。
    case 内的任何异常被捕获，记录到 EvalCaseResult.error，不中断批量执行。

    Usage::

        runner = EvalRunner(agent)
        cases = [
            EvalCase(
                name="greeting",
                messages=[Message(role="user", content="Hello")],
                assertion=lambda r: "hello" in r.turns[-1].response.content.lower(),
            )
        ]
        report = await runner.run(cases)
        print(f"Pass rate: {report.pass_rate:.0%}")
    """

    def __init__(self, agent: "NeoAgent") -> None:
        self._agent = agent

    async def run(self, cases: list[EvalCase]) -> EvalReport:
        """顺序执行所有 case，收集结果，输出 EvalReport。

        异常不中断批量执行：任何 case 内的异常被捕获，
        记录到 EvalCaseResult.error，passed=False。
        """
        from neoagent.eval.metrics import MetricsCollector
        from neoagent.events import (
            ProviderRequestEvent, ProviderResponseEvent,
            ToolCallEvent, TurnCompleteEvent,
        )

        results: list[EvalCaseResult] = []

        for case in cases:
            # 为本 case 创建独立 MetricsCollector，并注册到 EventBus
            collector = MetricsCollector()
            self._agent.event_bus.subscribe(ProviderRequestEvent, collector._on_provider_request)
            self._agent.event_bus.subscribe(ProviderResponseEvent, collector._on_provider_response)
            self._agent.event_bus.subscribe(ToolCallEvent, collector._on_tool_call)
            self._agent.event_bus.subscribe(TurnCompleteEvent, collector._on_turn_complete)

            try:
                result = await self._agent.run(
                    messages=list(case.messages),
                    max_turns=case.max_turns,
                )
                # 执行断言
                try:
                    passed = bool(case.assertion(result))
                    results.append(EvalCaseResult(
                        name=case.name,
                        passed=passed,
                        result=result,
                        error=None,
                        metrics=collector.get_session_metrics(),
                    ))
                except Exception as assert_exc:
                    results.append(EvalCaseResult(
                        name=case.name,
                        passed=False,
                        result=result,
                        error=f"assertion raised: {assert_exc}",
                        metrics=collector.get_session_metrics(),
                    ))

            except Exception as exc:
                results.append(EvalCaseResult(
                    name=case.name,
                    passed=False,
                    result=None,
                    error=str(exc),
                    metrics=collector.get_session_metrics(),
                ))

            finally:
                # 无论成功/失败，解除订阅避免下一 case 的数据污染
                self._agent.event_bus.unsubscribe(ProviderRequestEvent, collector._on_provider_request)
                self._agent.event_bus.unsubscribe(ProviderResponseEvent, collector._on_provider_response)
                self._agent.event_bus.unsubscribe(ToolCallEvent, collector._on_tool_call)
                self._agent.event_bus.unsubscribe(TurnCompleteEvent, collector._on_turn_complete)

        passed_count = sum(1 for r in results if r.passed)
        return EvalReport(
            total=len(results),
            passed=passed_count,
            failed=len(results) - passed_count,
            cases=results,
        )
```

## 附录：关键 API 约束（来自源码实测）

1. **`ToolCallEvent`** 字段为 `name: str, input_data: dict, call_id: str`（无 `turn_index`）—— MetricsCollector 中用 `max(self._pending.keys())` 归入最新轮。
2. **`TurnCompleteEvent`** 字段为 `turn_index: int, stop_reason: str, tool_call_count: int`。
3. **`ProviderResponseEvent`** 字段为 `content: tuple, stop_reason: str, input_tokens: int, output_tokens: int, turn: int`。
4. **`Session._storage`** 是 `field(default=None, repr=False)`——repr 关闭避免打印时触发存储调用；dataclass `__init__` 仍接受此参数，但 `Session.create()` 和 `_session_from_dict()` 不传此参数（保持默认 None）。
5. **`agent._provider.model`** 在 `AnthropicProvider`（第 41 行 `self.model = model`）和 `OpenAIProvider`（第 125 行 `self.model = model`）均直接暴露。
6. **`QueryLoop.run()`** 中 `_session` 是局部变量（不是 `self._session`），在循环体外创建后贯穿整个 run 执行，可直接调用 `_session.save_if_storage()`。
7. **`NeoAgent.run()`** 原来没有 `_temp` 变量，Task 3 需要新增 `_temp = session is None` 来区分临时 session（不做 auto-save）和用户传入 session（需要 bind_storage + final save）。
