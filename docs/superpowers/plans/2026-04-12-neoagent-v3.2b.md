**Goal:** Add Multi-Agent capability (Orchestrator + Workers) to neoagent SDK

**Architecture:** Independent Orchestrator class composing NeoAgent as decision engine, with WorkerPool for static worker management and TaskTracker for task lifecycle. Workers are independent NeoAgent instances created from WorkerCard definitions (.md frontmatter format). Five built-in tools (spawn_worker, delegate_task, list_workers, cancel_task, list_tasks) expose orchestration to the LLM.

**Tech Stack:** Python 3.11+, asyncio, pytest, dataclasses, PyYAML (for .md frontmatter parsing)

```python
from __future__ import annotations

import uuid
from dataclasses import dataclass, field
from typing import Literal, TYPE_CHECKING

if TYPE_CHECKING:
    from neoagent.core.types import Message


_DEFAULT_CONSTRAINTS: dict = {
    "max_turns": 20,
    "timeout": 1800,
    "max_output_tokens": 2000,
}


@dataclass(frozen=True)
class TokenUsage:
    """Token consumption summary for a task."""
    input_tokens: int
    output_tokens: int

    @property
    def total(self) -> int:
        return self.input_tokens + self.output_tokens


@dataclass(frozen=True)
class Task:
    """Envelope sent from Orchestrator to Worker."""
    task_id: str
    instruction: str
    context: tuple = ()           # tuple[Message, ...]
    constraints: dict = field(default_factory=lambda: dict(_DEFAULT_CONSTRAINTS))
    metadata: dict = field(default_factory=dict)

    @classmethod
    def create(
        cls,
        instruction: str,
        context: tuple = (),
        constraints: dict | None = None,
        metadata: dict | None = None,
    ) -> "Task":
        merged = dict(_DEFAULT_CONSTRAINTS)
        if constraints:
            merged.update(constraints)
        return cls(
            task_id=str(uuid.uuid4()),
            instruction=instruction,
            context=context,
            constraints=merged,
            metadata=metadata or {},
        )


@dataclass(frozen=True)
class TaskResult:
    """Result returned from Worker to Orchestrator."""
    task_id: str
    status: Literal["completed", "failed", "cancelled"]
    output: str
    error: str | None
    usage: TokenUsage | None
    work_summary: str | None    # programmatic summary when cancelled
    turns_completed: int

    @property
    def is_success(self) -> bool:
        return self.status == "completed"
```

```python
from __future__ import annotations

import logging
from pathlib import Path

import yaml

from neoagent.multi.worker import WorkerCard

logger = logging.getLogger(__name__)

_FRONTMATTER_SEP = "---"


def parse_worker_md(content: str) -> WorkerCard:
    """Parse a WorkerCard .md string (YAML frontmatter + body) into a WorkerCard.

    Raises ValueError on missing frontmatter, missing required fields, or
    malformed YAML.
    """
    lines = content.split("\n")

    # Find opening --- marker (first non-empty line must be ---)
    start = -1
    for i, line in enumerate(lines):
        stripped = line.strip()
        if stripped == _FRONTMATTER_SEP:
            start = i
            break
        elif stripped:
            # Non-empty line before first --- means no frontmatter
            raise ValueError(
                f"No YAML frontmatter found: expected '---' but got {stripped!r}"
            )

    if start == -1:
        raise ValueError("No YAML frontmatter found: file is empty or has no '---' marker")

    # Find closing --- marker
    end = -1
    for i in range(start + 1, len(lines)):
        if lines[i].strip() == _FRONTMATTER_SEP:
            end = i
            break

    if end == -1:
        raise ValueError("Malformed frontmatter: opening '---' found but no closing '---'")

    # Parse YAML between markers
    yaml_text = "\n".join(lines[start + 1 : end])
    try:
        fm = yaml.safe_load(yaml_text) or {}
    except yaml.YAMLError as exc:
        raise ValueError(f"Malformed YAML frontmatter: {exc}") from exc

    if not isinstance(fm, dict):
        raise ValueError(f"Frontmatter must be a YAML mapping, got {type(fm).__name__}")

    # Validate required fields
    if "name" not in fm or not fm["name"]:
        raise ValueError("WorkerCard frontmatter missing required field: 'name'")

    # Extract optional fields
    name: str = str(fm["name"])
    description: str = str(fm.get("description", ""))
    model: str | None = str(fm["model"]) if fm.get("model") else None

    raw_tags = fm.get("tags") or []
    tags: tuple[str, ...] = tuple(str(t) for t in raw_tags)

    raw_tools = fm.get("tools") or []
    tools: tuple[str, ...] = tuple(str(t) for t in raw_tools)

    # Body: everything after the closing ---
    body_lines = lines[end + 1 :]
    instruction = "\n".join(body_lines).strip()

    return WorkerCard(
        name=name,
        description=description,
        instruction=instruction,
        tags=tags,
        model=model,
        tools=tools,
    )


def load_workers(directory: Path) -> list[WorkerCard]:
    """Load all WorkerCard .md files from a directory.

    Raises ValueError if any .md file fails to parse.
    Returns empty list if directory contains no .md files.
    """
    directory = Path(directory)
    cards: list[WorkerCard] = []
    for path in sorted(directory.glob("*.md")):
        content = path.read_text(encoding="utf-8")
        try:
            card = parse_worker_md(content)
        except ValueError as exc:
            raise ValueError(f"Failed to parse worker file {path}: {exc}") from exc
        cards.append(card)
    return cards
```

```python
        self._cards: dict[str, WorkerCard] = {}
        self._lock = threading.Lock()

    def register(self, card: WorkerCard) -> None:
        """Register a WorkerCard. Overwrites existing entry with same name."""
        with self._lock:
            if card.name in self._cards:
                _logger.warning(
                    "WorkerPool: overwriting existing worker %r with new definition",
                    card.name,
                )
            self._cards[card.name] = card

    def find(self, name: str) -> WorkerCard | None:
        """Return WorkerCard by name, or None if not found."""
        with self._lock:
            return self._cards.get(name)

    def list_all(self) -> list[WorkerCard]:
        """Return a copy of all registered WorkerCards."""
        with self._lock:
            return list(self._cards.values())

    def remove(self, name: str) -> bool:
        """Remove a WorkerCard by name. Returns True if removed, False if not found."""
        with self._lock:
            if name in self._cards:
                del self._cards[name]
                return True
            return False

    @property
    def size(self) -> int:
        with self._lock:
            return len(self._cards)
```

```python
import asyncio
import logging
import threading
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from neoagent.core.types import Message, ToolUseBlock, TextBlock

_tracker_logger = logging.getLogger(__name__)


def _truncate(text: str, max_len: int) -> str:
    """Truncate a string to max_len chars, appending '...' if cut."""
    if len(text) <= max_len:
        return text
    return text[:max_len] + "..."


def _extract_work_summary(messages: list) -> str:
    """Deterministically extract a work summary from message history.
    No LLM calls — pure programmatic extraction.

    Args:
        messages: list[Message] from agent session
    """
    from neoagent.core.types import ToolUseBlock, TextBlock

    tool_calls: list[str] = []
    last_assistant_text: str | None = None

    for msg in messages:
        if msg.role != "assistant":
            continue

        content = msg.content
        if isinstance(content, str):
            last_assistant_text = content
            continue

        # Structured content blocks
        for block in content:
            if isinstance(block, ToolUseBlock):
                input_repr = _truncate(str(block.input), 50)
                tool_calls.append(f"{block.name}({input_repr})")
            elif isinstance(block, TextBlock) and block.text.strip():
                last_assistant_text = block.text

    parts: list[str] = []
    if tool_calls:
        recent = tool_calls[-5:]  # Last 5 tool calls only
        parts.append(f"已执行工具: {', '.join(recent)}")
    if last_assistant_text:
        parts.append(f"最后输出: {_truncate(last_assistant_text, 200)}")

    return "\n".join(parts) if parts else "（无有效工作摘要）"


class TaskTracker:
    """Tracks task lifecycle: pending → active → completed/cancelled/failed.

    Stores asyncio.Task references for cancellation.
    Thread-safe internal storage.
    """

    def __init__(self) -> None:
        self._tasks: dict[str, Task] = {}
        self._results: dict[str, TaskResult] = {}
        self._running: dict[str, asyncio.Task] = {}  # type: ignore[type-arg]
        self._lock = threading.Lock()

    def track(self, task: Task) -> None:
        """Register a Task as active (pending execution)."""
        with self._lock:
            self._tasks[task.task_id] = task

    def set_asyncio_task(self, task_id: str, asyncio_task: "asyncio.Task") -> None:
        """Associate an asyncio.Task with a tracked Task for cancellation."""
        with self._lock:
            self._running[task_id] = asyncio_task

    def complete(self, task_id: str, result: TaskResult) -> None:
        """Mark a task as complete and store its result."""
        with self._lock:
            self._results[task_id] = result
            self._running.pop(task_id, None)

    def get_task(self, task_id: str) -> "Task | None":
        with self._lock:
            return self._tasks.get(task_id)

    def get_result(self, task_id: str) -> "TaskResult | None":
        with self._lock:
            return self._results.get(task_id)

    def list_active(self) -> list[str]:
        """Return task_ids for tasks that have not yet completed."""
        with self._lock:
            completed = set(self._results.keys())
            return [tid for tid in self._tasks if tid not in completed]

    def list_all(self) -> list[dict]:
        """Return summary dicts for all tracked tasks."""
        with self._lock:
            result = []
            for tid, task in self._tasks.items():
                r = self._results.get(tid)
                result.append({
                    "task_id": tid,
                    "instruction": _truncate(task.instruction, 80),
                    "status": r.status if r else "active",
                })
            return result

    def cancel(self, task_id: str) -> bool:
        """Cancel a running task. Returns True if cancellation was requested."""
        with self._lock:
            # Cannot cancel if already completed
            if task_id in self._results:
                return False
            # Cannot cancel if not tracked
            if task_id not in self._tasks:
                return False
            asyncio_task = self._running.get(task_id)

        if asyncio_task is not None:
            asyncio_task.cancel()
            return True

        # Tracked but no asyncio task yet — still valid cancel request
        _tracker_logger.warning(
            "TaskTracker.cancel(%r): task tracked but no asyncio.Task registered yet",
            task_id,
        )
        return True
```

```python
from __future__ import annotations

import logging
from typing import TYPE_CHECKING

from neoagent.events import Event, WorkerEvent

if TYPE_CHECKING:
    from neoagent.events import EventBus

logger = logging.getLogger(__name__)


def _setup_event_bubble(
    worker_agent,
    orchestrator,
    worker_name: str,
    task_id: str,
    depth: int,
) -> None:
    """Subscribe to a Worker's EventBus and re-publish all events to the
    Orchestrator's EventBus wrapped in WorkerEvent.

    This is a pure Python callback — events never enter the LLM context.
    The Orchestrator LLM only sees TaskResult (final output).

    Args:
        worker_agent: NeoAgent instance for the Worker (has .event_bus)
        orchestrator: Orchestrator instance (has ._event_bus)
        worker_name: Name of the worker (for WorkerEvent.worker_name)
        task_id: Current task id (for WorkerEvent.task_id)
        depth: Nesting depth of this worker (for WorkerEvent.depth)
    """
    orch_bus: EventBus = orchestrator._event_bus

    def bubble_handler(event: Event) -> None:
        wrapped = WorkerEvent(
            worker_name=worker_name,
            task_id=task_id,
            depth=depth,
            inner=event,
        )
        try:
            orch_bus.emit(wrapped)
        except Exception:
            logger.exception(
                "Event bubble failed for worker=%r task=%r event=%s",
                worker_name, task_id, type(event).__name__,
            )

    worker_agent.event_bus.subscribe_all(bubble_handler)
```

```python
from __future__ import annotations
import logging
from dataclasses import dataclass, field
from pathlib import Path
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from neoagent.agent import NeoAgent
    from neoagent.multi.orchestrator import Orchestrator

logger = logging.getLogger(__name__)


# ── WorkerCard ────────────────────────────────────────────────────────────────

@dataclass(frozen=True)
class WorkerCard:
    name: str
    description: str
    instruction: str
    tags: tuple[str, ...] = ()
    model: str | None = None
    tools: tuple[str, ...] = ()


# ── WorkerPool ────────────────────────────────────────────────────────────────

class WorkerPool:
    def __init__(self) -> None:
        self._cards: dict[str, WorkerCard] = {}

    def register(self, card: WorkerCard) -> None:
        if card.name in self._cards:
            logger.warning("WorkerPool: overwriting existing worker %r", card.name)
        self._cards[card.name] = card

    def find(self, name: str) -> WorkerCard | None:
        return self._cards.get(name)

    def list_all(self) -> list[WorkerCard]:
        return list(self._cards.values())

    def remove(self, name: str) -> bool:
        if name in self._cards:
            del self._cards[name]
            return True
        return False


# ── Worker factory ────────────────────────────────────────────────────────────

def _create_worker_agent(
    card: WorkerCard,
    orchestrator: "Orchestrator",
    depth: int,
) -> "NeoAgent":
    """Create an independent NeoAgent instance from a WorkerCard.

    Registers only tools explicitly authorized by card.tools.
    Registers spawn_worker only if depth < orchestrator.max_depth.
    Sets up event bubbling to the Orchestrator's EventBus.
    """
    from neoagent.agent import NeoAgent
    from neoagent.config import NeoAgentConfig

    config = NeoAgentConfig(
        api_key=orchestrator.config.api_key,
        model=card.model or orchestrator.config.model,
        provider=orchestrator.config.provider,
        base_url=orchestrator.config.base_url,
        system_prompt=card.instruction,
        auto_approve_tools=True,  # internal workers auto-approve
    )
    agent = NeoAgent(config)

    # Register only tools authorized by WorkerCard
    for tool_name in card.tools:
        tool = orchestrator._tool_pool.get(tool_name)
        if tool is not None:
            agent.register_tool(tool)
        else:
            logger.debug(
                "_create_worker_agent: tool %r not found in pool, skipping", tool_name
            )

    # Depth guard: only register spawn_worker if depth < max_depth
    if depth < orchestrator.max_depth:
        from neoagent.multi.tools.spawn_worker import SpawnWorkerTool
        spawn_tool = SpawnWorkerTool(orchestrator=orchestrator, depth=depth)
        agent.register_tool(spawn_tool)

    # Event bubbling: worker events → orchestrator event bus
    from neoagent.multi.events import _setup_event_bubble
    _setup_event_bubble(agent, orchestrator, card.name, depth)

    return agent
```

```python
    """Execute a worker task. Handles timeout and cancellation gracefully."""
    timeout = task.constraints.get("timeout", 1800)
    max_turns = task.constraints.get("max_turns", 20)

    from neoagent.core.types import Message as _Message

    # Build message list: context + instruction
    messages = list(task.context) + [_Message(role="user", content=task.instruction)]

    try:
        async with asyncio.timeout(timeout):
            result = await agent.run(messages=messages)

        # Extract output from last assistant turn
        from neoagent.core.types import TextBlock as _TB
        output = ""
        if result.turns:
            last = result.turns[-1].response
            if isinstance(last.content, list):
                output = "\n".join(
                    b.text for b in last.content if isinstance(b, _TB)
                )
            elif isinstance(last.content, str):
                output = last.content

        # Truncate output to max_output_tokens (approximate: 4 chars/token)
        max_chars = task.max_output_tokens * 4
        if len(output) > max_chars:
            output = output[:max_chars] + "\n[truncated]"

        # Accumulate token usage across all turns
        total_input = sum(
            getattr(t.response, "input_tokens", 0) for t in result.turns
        )
        total_output = sum(
            getattr(t.response, "output_tokens", 0) for t in result.turns
        )
        # Prefer provider-level usage from session state
        usage = TokenUsage(input_tokens=total_input, output_tokens=total_output)

        return TaskResult(
            task_id=task.task_id,
            status="completed",
            output=output,
            error=None,
            usage=usage,
            work_summary=None,
            turns_completed=len(result.turns),
        )

    except asyncio.CancelledError:
        summary = _extract_work_summary(agent)
        return TaskResult(
            task_id=task.task_id,
            status="cancelled",
            output="",
            error=None,
            usage=None,
            work_summary=summary,
            turns_completed=0,
        )
    except Exception as e:
        return TaskResult(
            task_id=task.task_id,
            status="failed",
            output="",
            error=str(e),
            usage=None,
            work_summary=None,
```

```python
class SpawnWorkerTool(BaseTool):
    """Dynamically spawn a temporary Worker from a markdown definition and run a task.

    The WorkerCard is NOT registered to WorkerPool — it is ephemeral.
    Parses md_definition → WorkerCard → NeoAgent → runs task → returns TaskResult.
    """
    name: str = "spawn_worker"
    description: str = (
        "Dynamically create a temporary Worker from a markdown definition and run a task. "
        "The md_definition must be a valid WorkerCard .md format with a 'name' field. "
        "The worker is ephemeral — not added to the worker pool."
    )
    input_model: type[BaseModel] = SpawnWorkerInput
    permission: str = "auto"

    def __init__(self, orchestrator: "Orchestrator", depth: int) -> None:
        self._orchestrator = orchestrator
        self._depth = depth

    async def execute(self, input: BaseModel) -> ToolResult:
        assert isinstance(input, SpawnWorkerInput)

        # 1. Parse md_definition → WorkerCard
        from neoagent.multi.parser import parse_worker_md
        from neoagent.multi.task import Task, _run_worker
        from neoagent.multi.worker import _create_worker_agent

        try:
            card = parse_worker_md(input.md_definition)
        except ValueError as e:
            return ToolResult(
                call_id="",
                output=f"Failed to parse worker definition: {e}",
                is_error=True,
            )

        # 2. Create worker agent (depth + 1 for spawned workers)
        task_id = str(uuid.uuid4())[:8]
        worker_agent = _create_worker_agent(card, self._orchestrator, self._depth + 1)

        # 3. Build Task
        task = Task(
            task_id=task_id,
            instruction=input.instruction,
        )

        # 4. Execute under semaphore
        async with self._orchestrator._semaphore:
            result = await _run_worker(worker_agent, task, self._orchestrator._task_tracker)

        # 5. Track completion
```

```python
from __future__ import annotations
import asyncio
import logging
from pathlib import Path
from typing import TYPE_CHECKING

from neoagent.config import NeoAgentConfig
from neoagent.agent import NeoAgent
from neoagent.multi.worker import WorkerPool
from neoagent.multi.task import TaskTracker
from neoagent.tools.base import BaseTool

if TYPE_CHECKING:
    pass

logger = logging.getLogger(__name__)


def _create_provider(config: NeoAgentConfig):
    """Re-export to allow patching in tests."""
    from neoagent.agent import _create_provider as _cp
    return _cp(config)


class Orchestrator:
    """Multi-agent coordinator.

    Composes a NeoAgent (brain) with WorkerPool and TaskTracker.
    The brain's LLM has access to 5 internal tools for spawning and delegating tasks.
    Workers are independent NeoAgent instances created on demand.

    Usage::

        orch = Orchestrator(NeoAgentConfig(api_key="..."))
        orch.register_worker(WorkerCard(name="reviewer", ...))
        orch.register_tool(ReadTool())
        result = await orch.run("Review these three files for security issues")
    """

    def __init__(
        self,
        config: NeoAgentConfig,
        max_depth: int = 2,
        max_concurrent_workers: int = 5,
    ) -> None:
        self.config = config
        self.max_depth = max_depth
        self.max_concurrent_workers = max_concurrent_workers

        # Internal components
        self._worker_pool = WorkerPool()
        self._task_tracker = TaskTracker()
        self._tool_pool: dict[str, BaseTool] = {}
        self._semaphore = asyncio.Semaphore(max_concurrent_workers)

        # Brain: the orchestrator's own NeoAgent
        self._brain = NeoAgent(config)

        # Register 5 internal tools onto the brain
        self._register_internal_tools()

    def _register_internal_tools(self) -> None:
        from neoagent.multi.tools.spawn_worker import SpawnWorkerTool
        from neoagent.multi.tools.delegate_task import DelegateTaskTool
        from neoagent.multi.tools.cancel_task import CancelTaskTool
        from neoagent.multi.tools.list_tools import ListWorkersTool, ListTasksTool

        self._brain.register_tool(SpawnWorkerTool(orchestrator=self, depth=0))
        self._brain.register_tool(DelegateTaskTool(orchestrator=self, depth=0))
        self._brain.register_tool(CancelTaskTool(orchestrator=self))
        self._brain.register_tool(ListWorkersTool(orchestrator=self))
        self._brain.register_tool(ListTasksTool(orchestrator=self))

    # ── Worker registration ───────────────────────────────────────────────────

    def register_worker(self, card) -> None:
        """Register a static WorkerCard by reference."""
        self._worker_pool.register(card)

    def load_workers(self, directory: str | Path) -> None:
        """Load all .md WorkerCard files from a directory and register them."""
        from neoagent.multi.parser import load_workers
        for card in load_workers(Path(directory)):
            self._worker_pool.register(card)

    # ── Tool pool management ──────────────────────────────────────────────────

    def register_tool(self, tool: BaseTool) -> None:
        """Add a tool to the global pool available for Worker authorization."""
        self._tool_pool[tool.name] = tool

    # ── Execution ─────────────────────────────────────────────────────────────

    async def run(self, message: str) -> str:
        """Run the orchestrator with a user message.

        The brain's LLM decides how to decompose and delegate the work.
        Returns the final response string.
        """
        return await self._brain.chat(message)

    # ── Lifecycle ─────────────────────────────────────────────────────────────

    async def close(self) -> None:
        """Cancel all running worker tasks and clean up the brain."""
        # Cancel all tracked active tasks
        active_ids = self._task_tracker.list_active()
        for task_id in active_ids:
            await self._task_tracker.cancel(task_id)

        # Close brain (MCP connections etc.)
        await self._brain.close()

    async def __aenter__(self) -> "Orchestrator":
        return self

    async def __aexit__(self, *_) -> None:
        await self.close()
```

```python
class TaskTracker:
    def __init__(self) -> None:
        self._tasks: dict[str, asyncio.Task] = {}       # task_id → asyncio.Task
        self._results: dict[str, TaskResult] = {}       # task_id → TaskResult
        self._metadata: dict[str, dict] = {}            # task_id → {worker_name, status}

    def track(self, task: Task, async_task: asyncio.Task, worker_name: str = "") -> None:
        """Register a running asyncio.Task for tracking."""
        self._tasks[task.task_id] = async_task
        self._metadata[task.task_id] = {
            "task_id": task.task_id,
            "worker_name": worker_name,
            "status": "running",
        }

    def complete(self, task_id: str, result: TaskResult) -> None:
        """Mark task as complete and store result."""
        self._results[task_id] = result
        self._tasks.pop(task_id, None)
        if task_id in self._metadata:
            self._metadata[task_id]["status"] = result.status

    async def cancel(self, task_id: str) -> bool:
        """Cancel a running task. Returns True if found and cancelled."""
        async_task = self._tasks.get(task_id)
        if async_task is None:
            return False
        async_task.cancel()
        try:
            await asyncio.wait_for(asyncio.shield(async_task), timeout=5.0)
        except (asyncio.CancelledError, asyncio.TimeoutError):
            pass
        self._tasks.pop(task_id, None)
        if task_id in self._metadata:
            self._metadata[task_id]["status"] = "cancelled"
        return True

    def get_result(self, task_id: str) -> TaskResult | None:
        return self._results.get(task_id)

    def list_active(self) -> list[str]:
        return [tid for tid, t in self._tasks.items() if not t.done()]

    def list_all(self) -> list[dict]:
        result = []
        for tid, meta in self._metadata.items():
            result.append(dict(meta))
        return result
```

```python
def _emit_dispatch_event(orchestrator, task_id: str, worker_name: str, instruction: str, depth: int) -> None:
    """Emit TaskDispatchEvent on orchestrator's event bus."""
    from neoagent.multi.events import TaskDispatchEvent
```

**Key design choices in Part 2:**

- `_create_worker_agent` is a pure factory function (not a method), matching the spec's design and making it easy to mock in tests
- `system_prompt` is added to `NeoAgentConfig` as `None`-default field — backward compatible, no existing code breaks
- `_run_worker` lives in `task.py` (close to `Task`/`TaskResult` data types), not in a tool file — clean separation
- `_format_task_result` is a shared helper in `tools/__init__.py` — avoids duplication between `spawn_worker` and `delegate_task`
- Event bubbling uses `subscribe_all` on the worker's `EventBus` — wraps every event as `WorkerEvent` without needing to know event types upfront
- `_ALL_EVENT_TYPES` uses lazy import to avoid `events.py` → `multi/events.py` circular dependency
- `CancelTaskTool.execute` is synchronous-friendly — `async def execute` calls `await tracker.cancel()` which handles asyncio task cancellation
- Integration tests mock LLM calls (`agent.run = AsyncMock`) but use real `WorkerPool`, `TaskTracker`, `EventBus`, `Semaphore` — tests the wiring, not the LLM
